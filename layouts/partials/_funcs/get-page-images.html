{{- /* Overrides Hugo's embedded partial behind the opengraph and twitter_cards templates, so every post gets a social image without front matter: 1. the first entry of front matter `images`, else 2. the first page-bundle image used in the post body, else 3. the site-wide `params.images`. Bundle images wider than 1200px are resized; SVG is skipped (social platforms don't render it). */ -}} {{- $imgs := slice -}} {{- $res := "" -}} {{- $url := "" -}} {{- with .Params.images -}} {{- $first := index (slice | append .) 0 -}} {{- with $.Resources.Get $first -}}{{- $res = . -}}{{- else -}}{{- $url = $first -}}{{- end -}} {{- else -}} {{- range findRESubmatch `!\[[^\]]*\]\(([^)\s]+)` .RawContent -}} {{- if not $res -}} {{- with $.Resources.Get (index . 1) -}} {{- if and (eq .MediaType.MainType "image") (ne .MediaType.SubType "svg") -}}{{- $res = . -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $res -}} {{- if gt .Width 1200 -}}{{- $res = .Resize "1200x" -}}{{- end -}} {{- $imgs = $imgs | append (dict "Image" $res "RelPermalink" $res.RelPermalink "Permalink" $res.Permalink) -}} {{- else -}} {{- if not $url -}} {{- with site.Params.images -}}{{- $url = index (slice | append .) 0 -}}{{- end -}} {{- end -}} {{- with $url -}} {{- $u := absURL . -}} {{- $imgs = $imgs | append (dict "Image" "" "RelPermalink" $u "Permalink" $u) -}} {{- end -}} {{- end -}} {{- return $imgs -}}
