{{- /* The theme's prebuilt theme.css, adjusted at build time (WEBS-30): - Font Awesome's @font-face rules use font-display:block; swap them so text and icons aren't invisible while the icon fonts load. - The text fonts' woff2 files point to latin/latin-ext subsets in static/fonts/subset/ (Greek, Cyrillic and Vietnamese glyphs dropped, about half the size). The other formats stay as fallbacks. - Font Awesome's fonts load from ./fonts/fa/ (mounted in hugo.toml), not ./fonts/_vendor/@fortawesome/..., which Cloudflare redirects in a loop (WEBS-31). - Colours darkened to pass WCAG AA contrast (WEBS-31, see below). Fingerprinted, so it can be cached; published at the site root so the relative ./fonts/ URLs still resolve. theme.css is mounted from the theme's static/ into assets/ (hugo.toml). */ -}} {{- $css := resources.Get "bilberry/theme.css" -}} {{- $s := replace $css.Content "font-display:block" "font-display:swap" -}} {{- $s = replaceRE `\./fonts/([a-z0-9_-]+)\.woff2\?[0-9a-f]+` "./fonts/subset/$1.woff2" $s -}} {{- $s = replace $s "./fonts/_vendor/@fortawesome/fontawesome-free/" "./fonts/fa/" -}} {{- /* WEBS-31: WCAG AA contrast. The accent green (#5c8b59, 3.97:1 on white) becomes #4a7347 (5.48:1); post dates (#999) and the credits bar (#777 on #f1f1f1) get darker greys. */ -}} {{- $s = replace $s "#5c8b59" "#4a7347" -}} {{- $s = print $s ".bilberry-hugo-theme article .content .meta{color:#6b6b6b}.bilberry-hugo-theme .credits .container{color:#666}" -}} {{- return (resources.FromString "theme.css" $s | fingerprint) -}}
