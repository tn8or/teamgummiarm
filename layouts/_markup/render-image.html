{{- /* Page-bundle raster images are resized (WebP, srcset) so posts don't load the multi-MB original; the image links to the original for full size. Paths are resolved to permalinks so they also work where the post body is rendered on another page (front page, lists). Remote images, static files, SVG and GIF render as a plain img. */ -}} {{- $src := .Destination -}} {{- $res := "" -}} {{- if and (not (urls.Parse $src).Scheme) (not (hasPrefix $src "/")) -}} {{- $res = .PageInner.Resources.Get $src -}} {{- end -}} {{- if and $res (eq $res.MediaType.MainType "image") (not (in (slice "svg" "gif") $res.MediaType.SubType)) -}} {{- $srcset := slice -}} {{- $fallback := "" -}} {{- range $w := slice 480 800 1200 1600 -}} {{- if lt $w $res.Width -}} {{- $img := $res.Resize (printf "%dx webp q80" $w) -}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink $w) -}} {{- if eq $w 800 -}}{{- $fallback = $img -}}{{- end -}} {{- end -}} {{- end -}} {{- if not $fallback -}} {{- $fallback = $res.Resize (printf "%dx webp q80" $res.Width) -}} {{- $srcset = $srcset | append (printf "%s %dw" $fallback.RelPermalink $res.Width) -}} {{- end -}} {{- else -}} {{- with $res -}}{{- $src = .RelPermalink -}}{{- end -}} {{- end -}}
